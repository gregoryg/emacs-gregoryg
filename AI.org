** An experiment in composable prompts using presets
    Let's make a list of all directories that prompts may be found in - note that this is a subset of the =gptel-prompts= package by @jwiegley that I only discovered today [2025-07-03 Thu], but it suits my needs.

    We have a function that will work with 2 streams in my AIPIHKAL project
    #+begin_src emacs-lisp :tangle no
      (defcustom gjg/prompt-segment-directories
        '("~/projects/ai/AIPIHKAL/composable-prompts/" "~/projects/ai/AIPIHKAL/system-prompts")
        "A list of directories to search for gptel prompt segments.
      The search is performed in the order of the directories in this list."
        :type '(repeat string)
        :group 'gptel)

      (defun gjg/find-prompt-file (slug)
        "Find the full path for a prompt segment SLUG.

      Searches in =gjg/prompt-segment-directories= for a file that
      matches the SLUG. The filename may be prefixed with numbers (e.g.,
      \"00-\") and have an extension of .org, .md, or .txt.

      For example, the slug \"roles\" could match
      \"~/path/to/prompts/00-roles.org\".

      If multiple files match, a warning is issued and the first found
      file is returned."
        ;; Use =cl-loop= to concisely search across multiple directories.
        (let ((found-files
               (cl-loop for dir in gjg/prompt-segment-directories
                        for expanded-dir = (expand-file-name dir)
                        when (file-directory-p expanded-dir)
                        nconc (cl-loop for file in (directory-files expanded-dir)
                                       for full-path = (expand-file-name file expanded-dir)
                                       ;; Construct a regexp to find files like '00-slug.org'
                                       when (string-match-p
                                             (format "\\(?:[0-9]+-\\)?%s\\.\\(org\\|md\\|txt\\)\\'"
                                                     (regexp-quote slug))
                                             file)
                                       collect full-path))))

          (when (> (length found-files) 1)
            (warn "Found multiple files for slug '%s': %s. Using the first one."
                  slug found-files))

          (car found-files)))

      (defun gjg/build-system-prompt (slugs)
        "Build a system prompt by concatenating files matching SLUGS.

      For each slug in the list SLUGS, this function finds a
      corresponding prompt file using =gjg/find-prompt-file=,
      reads its content, and concatenates all contents together,
      separated by double newlines for clarity."
        (let* ((files (mapcar #'gjg/find-prompt-file slugs))
               (valid-files (cl-remove-if-not #'identity files))
               (contents (mapcar #'f-read-text valid-files)))
          (mapconcat #'identity contents "\n\n")))

**** Add a very long buffer-specific timout for mcp.el
     mcp.el/mcp-hub does not alter the default jsonrpc timeout of 10 seconds.  So if any mcp call takes more than 10 seconds to return, we end up with a hung tool call

     One area where this bites us in the collective arse is when using thinking LLMs in MCP calls (as with mcp-rubber-duck).

     So ... let's make a buffer-local substantial modification to the global default timeout of 10 seconds!

                 (lambda ()
                   (setq-local jsonrpc-default-request-timeout 600))) ;; 10 minutes, just to be safe

               (sentence-end "\\([.?!
        ]\\)"))

          ;; (with-temp-buffer
          (with-current-buffer (get-buffer-create (concat "*" temp-name "*"))

    #+begin_src emacs-lisp
      (require 'json)
      (require 'url)
      (require 'url-http)
      (require 'subr-x)

      (defun gjg/generate-image--extract-data-url (response)
        "Extract a base64 image data URL from OpenRouter RESPONSE.

      RESPONSE is a plist parsed from JSON.  Return the first image data
      URL found, or nil if none was present."
        (or (map-nested-elt response '(:choices 0 :message :images 0 :image_url :url))
            (map-nested-elt response '(:choices 0 :message :images 0 :imageUrl :url))
            (let ((b64-json (map-nested-elt response '(:data 0 :b64_json))))
              (when (stringp b64-json)
                (concat "data:image/png;base64," b64-json)))))

      (defun gjg/generate-image--decode-data-url (data-url)
        "Decode image bytes from DATA-URL.

      DATA-URL should be a string of the form
      =data:image/png;base64,...=.  Return the decoded binary string, or
      nil if DATA-URL is malformed."
        (when (and (stringp data-url)
                   (string-match "\\`data:image/[^;]+;base64,\\(.+\\)\\'" data-url))
          (base64-decode-string (match-string 1 data-url))))

      (defun gjg/generate-image--resolve-api-key ()
        "Resolve the API key for OpenRouter image generation."
        (let ((openrouter-backend
               (and (boundp 'gptel--known-backends)
                    (alist-get "openrouter" gptel--known-backends nil nil #'equal))))
          (or (getenv "OPENAI_API_KEY")
              (and openrouter-backend
                   (gptel--get-api-key (gptel-backend-key openrouter-backend)))
              (cond
               ((functionp gptel-api-key)
                (string-trim-right (funcall gptel-api-key) "[\n\r]+"))
               ((stringp gptel-api-key)
                (string-trim-right gptel-api-key "[\n\r]+"))
               ((and (symbolp gptel-api-key) (boundp gptel-api-key))
                (gptel--get-api-key (symbol-value gptel-api-key)))))))

      (defun gjg/generate-image--handle-response (status output-file)
        "Parse current buffer as an OpenRouter image response.

      STATUS is the `url-retrieve' status plist.  OUTPUT-FILE is where the
      PNG should be written.  Return a user-facing status string."
        (if (plist-get status :error)
            (format "Image generation failed: %s"
                    (mapconcat #'gptel--to-string
                               (plist-get status :error)
                               " "))
          (goto-char (point-min))
          (re-search-forward "^$" nil 'move)
          (forward-char)
          (let* ((response (gptel--json-read))
                 (error-info (plist-get response :error))
                 (data-url (gjg/generate-image--extract-data-url response))
                 (image-bytes (gjg/generate-image--decode-data-url data-url)))
            (when error-info
              (error "%s"
                     (or (plist-get error-info :message)
                         (prin1-to-string error-info))))
            (unless image-bytes
              (error "response did not contain image data"))
            (with-temp-file output-file
              (set-buffer-multibyte nil)
              (insert image-bytes))
            (format
             (concat
              "Success. Image saved to: %s\n"
              "Org link: [[file:%s][generated image]]\n"
              "Tell the user where it is, and if helpful include the Org link verbatim.")
             output-file output-file))))

      (defun gjg/generate-image-tool (callback prompt)
        "Generate an image from PROMPT using OpenRouter and save it locally.

      CALLBACK is the gptel async continuation.  Call it with a user-facing
      status string that includes the saved path and an Org mode file link."
        (let ((api-key (gjg/generate-image--resolve-api-key)))
          (if (not (and api-key (not (string-empty-p api-key))))
              (funcall callback
                       "Image generation failed: could not resolve OpenRouter API key")
            (let* ((url-request-method "POST")
                   (url-request-extra-headers
                    `(("Content-Type" . "application/json")
                      ("Authorization" . ,(concat "Bearer " api-key))))
                   (url-request-data
                    (gptel--json-encode
                     `(:model "openai/gpt-5-image-mini"
                       :messages [(:role "user" :content ,prompt)]
                       :modalities ["image" "text"]
                       :stream :json-false)))
                   (output-file
                    (expand-file-name
                     (format "img_%s.png" (format-time-string "%Y%m%d_%H%M%S"))
                     temporary-file-directory)))
              (url-retrieve
               "https://openrouter.ai/api/v1/chat/completions"
               (lambda (status)
                 (let ((result
                        (condition-case errdata
                            (gjg/generate-image--handle-response status output-file)
                          (error
                           (format "Image generation failed: %s"
                                   (error-message-string errdata))))))
                   (unwind-protect
                       (funcall callback result)
                     (kill-buffer (current-buffer)))))
               nil t)))))

      (gptel-make-tool
       :name "generate_image"
       :description "Generate an image using a text prompt and return the local path to the image file."
       :args (list '(:name "prompt"
                     :type string
                     :description "The detailed visual prompt for image generation"))
       :function #'gjg/generate-image-tool
       :async t
       :category "media")
